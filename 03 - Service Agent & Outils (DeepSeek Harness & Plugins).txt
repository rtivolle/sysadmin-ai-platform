03 - Service Agent & Outils : DeepSeek Harness (dsh) & Plugins

Rôle : Environnement d'exécution agentique, orchestration des boucles de raisonnement et boîte à outils d'administration système.

Technologies : DeepSeek Harness (dsh) + Micro-noyau Cordis + Plugins TypeScript sur mesure.

Licence : 100 % Open Source (MIT).

1. Fonctionnement de DeepSeek Harness et Cordis

DeepSeek Harness repose sur Cordis, un framework modulaire où chaque fonction (adaptateur LLM, bac à sable, outils, gestion de sessions) est un plugin injecté dans un contexte partagé (ctx.*). Les profils d'exécution permettent d'étendre ou de substituer n'importe quelle brique sans modifier le code source principal.

2. Fichier de Profil Cordis : profiles/sysadmin/cordis.patch.yml

Ce fichier configure l'agent pour utiliser le proxy LiteLLM local, remplace le bac à sable par défaut par Bubblewrap et charge les outils spécialisés pour administrateurs système :

# profiles/sysadmin/cordis.patch.yml- insert:    - id: llm-openai-litellm      name: '@deepseek-ai/dsh-llm-openai'      config:        baseURL: 'http://127.0.0.1:4000/v1'        apiKey: '${DSH_USER_API_KEY}'        defaultModel: 'heavy-model'        extraHeaders:          X-Dynamo-Session-Id: '${DSH_SESSION_ID}'          X-Dynamo-Priority: 'interactive'- replace:    - id: sandbox      name: 'dsh-plugin-sysadmin-sandbox'- insert:    - id: sysadmin-tools      name: 'dsh-plugin-sysadmin-tools'

3. Implémentation du Plugin d'Outils Sysadmin (TypeScript)

Enregistré sous plugins/dsh-plugin-sysadmin-tools/src/index.ts, ce plugin fournit les outils indispensables pour l'exploitation sans déborder la fenêtre de contexte :

import { Context } from 'cordis';import { spawn } from 'node:child_process';import * as fs from 'node:fs/promises';const DANGEROUS_COMMANDS = [/rm\s+-rf/i, /mkfs/i, /dd\s+if=/i, />\s*\/dev\/sd/i, /iptables\s+-F/i, /reboot/i];const MUTATING_COMMANDS = [/systemctl\s+(restart|stop|start)/i, /docker\s+(rm|kill|stop)/i, /sed\s+-i/i, /chown/i, /chmod/i];export default function apply(ctx: Context) {  // 1. Sas de sécurité Human-in-the-Loop (interception pré-exécution)  ctx.on('tools/pre-execute', async (call, next) => {    if (call.name === 'bash' || call.name === 'terminal') {      const command = call.arguments?.command as string;      // Blocage immédiat des commandes destructrices      if (DANGEROUS_COMMANDS.some((regex) => regex.test(command))) {        throw new Error(`Violation de sécurité : Commande interdite par politique de sécurité : "${command}"`);      }      // Demande d'approbation explicite pour les commandes modificatrices      if (MUTATING_COMMANDS.some((regex) => regex.test(command))) {        ctx.emit('session/event', {          type: 'approval_required',          level: 'critical',          command,          message: `L'agent demande à exécuter une opération modificatrice : "${command}". Confirmer ? (y/N)`,        });        const approved = await ctx.promptApproval(call.sessionId, command);        if (!approved) {          throw new Error("Opération refusée par l'administrateur système.");        }      }    }    return next();  });  // 2. Outil : Recherche ciblée dans les journaux (ripgrep & journalctl JSON)  ctx.tools?.register({    name: 'search_log_stream',    description: 'Recherche ciblée dans les logs volumineux via ripgrep ou journalctl JSON (max 50 occurrences, contexte ±2 lignes).',    parameters: {      type: 'object',      properties: {        target: { type: 'string', description: 'Chemin du fichier log ou unité systemd (ex: nginx.service)' },        pattern: { type: 'string', description: 'Motif regex de recherche' },        isJournalctl: { type: 'boolean', default: false },        maxMatches: { type: 'number', default: 50 },      },      required: ['target', 'pattern'],    },    async execute({ target, pattern, isJournalctl, maxMatches = 50 }) {      return new Promise((resolve) => {        let cmd = '/usr/bin/rg';        let args = ['--line-number', '--max-count', String(maxMatches), '-C', '2', pattern, target];        if (isJournalctl) {          cmd = '/usr/bin/journalctl';          args = ['-u', target, '-o', 'json', '--since', '1 hour ago', '--grep', pattern];        }        const proc = spawn(cmd, args);        let stdout = '', stderr = '';        proc.stdout.on('data', (d) => { stdout += d; });        proc.stderr.on('data', (d) => { stderr += d; });        proc.on('close', (code) => {          if (code !== 0 && stdout.length === 0) resolve({ matched: false, message: `Aucune correspondance ou erreur: ${stderr}` });          else resolve({ matched: true, output: stdout });        });      });    },  });  // 3. Outil : Validation syntaxique & Diff unifié  ctx.tools?.register({    name: 'config_lint_and_diff',    description: 'Vérifie la syntaxe (YAML/JSON/systemd) d\'une modification et produit un diff unifié avant application.',    parameters: {      type: 'object',      properties: {        targetFile: { type: 'string', description: 'Fichier cible de configuration' },        proposedContent: { type: 'string', description: 'Contenu proposé à tester' },      },      required: ['targetFile', 'proposedContent'],    },    async execute({ targetFile, proposedContent }) {      const original = await fs.readFile(targetFile, 'utf-8').catch(() => '');      if (targetFile.endsWith('.json')) {        try { JSON.parse(proposedContent); } catch (e: any) { return { valid: false, error: `Erreur syntaxe JSON: ${e.message}` }; }      }      return { valid: true, originalSize: original.length, proposedSize: proposedContent.length };    },  });  // 4. Outil : Lecture de runbooks et procédures internes  ctx.tools?.register({    name: 'doc_runbook_reader',    description: 'Extrait une section spécifique d\'un runbook Markdown d\'exploitation.',    parameters: {      type: 'object',      properties: {        runbookPath: { type: 'string', description: 'Chemin vers le fichier Markdown' },        sectionTitle: { type: 'string', description: 'Titre ou mot-clé de la section' },      },      required: ['runbookPath', 'sectionTitle'],    },    async execute({ runbookPath, sectionTitle }) {      const content = await fs.readFile(runbookPath, 'utf-8');      const lines = content.split('\n');      const extracted: string[] = [];      let capturing = false;      for (const line of lines) {        if (line.toLowerCase().includes(sectionTitle.toLowerCase())) capturing = true;        else if (capturing && line.startsWith('#')) break;        if (capturing) extracted.push(line);      }      return { found: extracted.length > 0, section: extracted.join('\n') || 'Section introuvable' };    },  });}
